📦 Express Integration Guide

const express = require('express');
const { createTrafficGuard } = require('./middleware/trafficGuard');
const RateLimiter = require('./engine/rateLimiter');
const AbuseDetector = require('./engine/abuseDetector');
const BlacklistManager = require('./engine/blacklistManager');

const app = express();
// Mount TrafficGuard across all API endpoints:
app.use(createTrafficGuard(rateLimiter, abuseDetector, blacklistManager));

app.get('/api/resource', (req, res) => {
  res.json({ secure: true, data: 'Protected by RateGuard Pro' });
});
